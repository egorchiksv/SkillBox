route - покажет таблицу маршрутизации
routel - покажет подобную таблицу маршрутизации
ip route - также выведет информацию по маршрутизации
sudo ip route add default via 10.0.2.14 - команда пропишет маршрут по умолчанию для указанного ip. Маршрут будет активен до перезагрузки.
Для постоянной настройки шлюза по умолчанию, необходимо его прописать в файл в каталоге /etc/network/interfaces.d
ip route show - покажет настроенные маршруты
sudo ip route add 10.10.3.0/24 dev lo - добавляем маршрут через локальный интерфейс
sudo ip route delete 10.10.3.0/24 dev lo - удаление маршрута
sudo route add default gw 10.10.1.20 - добавляем шлюз

ifconfig - покажет все доступные интерфейсы
ifconfig eth0 nodeone netmask 255.0.0.0 - определяет маршрут в сети через интерфейс eth0

sudo apt install - установка пакетов
sudo apt-get update - получение обновлений
apt install openssh-server - установка ssh
sudo systemctl restart ssh - перезапуск службы
systemctl status ssh - проверка статуса службы ssh
sudo wfw disable - выключение брендмауэра
ssh kali@10.0.2.15 - подключение по ssh
ssh kali@10.0.2.15 -p 45- подключение по ssh в случае, если порт нестандартный
sudo apt-get install vsftpd - установка службы для подключения
cp /etc/vsftpd.conf - сделает копию файла конфигурации службы
sudo adduser admin - создадим нового пользователя
sudo mkdir /home/admin/ftp - создадим каталог для содержания файлов передаваемых по ftp
sudo chown nobody:nogroup /home/admin/ftp - назначим владельца для папки
sudo chmod a-w /home/admin/ftp - запретим запись в папку
sudo ls -la /home/admin/ftp - проверим установленые права
sudo mkdir /home/admin/ftp/files - создадим папку для загрузки файлов
sudo chown admin:admin /home/admin/ftp/files - назначим владельца admin папки files 
sudo ls -la /home/admin/ftp/files - проверим также владельца каталога
sudo nano /etc/vsftpd.conf - открыть конфигурационный службы
sudo systemctl restart vsftpd - перезапуск службы
ftp -p 10.0.0.5 - подключение к ftp

sudo apt-get install sendmail - установка почтовой службы
nano /etc/php/8.4/apache2/php.ini - открытие конфига (в строке раскоментировать строку sendmail_from и после = вписать /usr/sbin/sendmail -t -l)
sudo service apache2 restart - перезапуск почтовой службы

sudo apt-get install nfs-kernel-server - установка nfs на сервер (сетевая файловая система)
rpcinfo -p | grep nfs - проверить правильно ли настроены порты у службы nfs
cat /proc/filesystems| grep nfs - проверить поддрежку файловой системы ядром
sudo modprobe nfs - загрузка nfs в ядро, в случае, если оно не добавлено в файл /proc/filesystems
sudo apt-get install nfs-common - установка nfs на клиента
sudo mkdir /var/nfs - создание общей папки для доступа в сети
sudo nano /etc/exports - файл настройки доспупа к папкам системы nfs. В конец файла необходимо добавить путь к папке, которая создана для общего доступа /var/nfs 127.0.0.1(rw,sync)
sudo exportfs -a - команда для обновления таблицы экспорта после всех настроек
sudo mount 127.0.0.1:/var/nfs /mnt/ - подключение к созданной папке
sudo umount 127.0.0.1:/var/nfs /mnt/ - отклдчение папки

sudo apt-get install isc-dhcp-server - установка dhcp сервера
sudo nano /etc/default/isc-dhcp-server - открытие конфигурационного файла dhcp сервера. Добавляем команду для прослушки интерфейса INTERFACES="eth0"
sudo nano /etc/dhcp/dhcpd.conf - открытие оснонвного конфигурационного файла dhcp сервера

sudo apt-get install wpasupplicant wireless-tools - утилиты для настройки беспроводного соединения
iwlist wlan0 scan - команда для сканирования беспроводной сети
sudo nano /etc/network/interfaces - конфигурационный файл утилиты настройки бепроводоной сети
